board_stm32f103/
├─ cubemx_gen/               			# 【自动生成层】CubeMX专属隔离目录
│  ├─ Core/Inc + Src         			# MX_GPIO、MX_USART、系统时钟、中断初始化
│  └─ Drivers/STM32F1xx_HAL_Driver 	# 原厂HAL库底层
├─ Core/                     				# 【业务核心手写层】项目专属逻辑，完全独立
│  ├─ config/                				# 全局宏、引脚、任务参数配置
│  ├─ tasks/                 				# FreeRTOS所有任务入口文件
│  └─ app/                   				# 设备联动业务逻辑（无硬件操作，纯逻辑判断）
├─ Drivers/                  				# 【扩展驱动层】手写外设封装（按键消抖、电机、舵机等）
├─ Middleware/                			# 【通用中间件层】全项目可复用工具组件
├─ FreeRTOS/                 				# RTOS内核与配置文件
├─ User/                     				# 【系统启动层】统一硬件+RTOS初始化入口
└─ Keil_Project/             				# IDE工程文件
